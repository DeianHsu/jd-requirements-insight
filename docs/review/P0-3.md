```powershell

$targetPath = 'scripts/experiments/p0_3/generate_validation_cases.py'

  --invert-paths --path $targetPath

if (-not (git remote)) {
  git remote add origin $originUrl
}

if (git log --all --oneline -- $targetPath) {
  throw '目标路径仍存在于历史中'
}

  throw '目标路径对象仍可达'
```
